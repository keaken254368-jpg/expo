import React, { useEffect, useMemo, useState } from "react";

/**
 * Motor Maintenance App (TH)
 * - รายชื่อมอเตอร์ 100 ตัว (M001..M100)
 * - ค้นหาชื่อ + ประวัติพร้อมกัน (Global Search)
 * - หน้ารายชื่อหาง่าย กรองได้แบบพิมพ์ทันที
 * - หน้ารายละเอียด: ข้อมูลมอเตอร์ + ประวัติซ่อม/บำรุง + รายการ PA
 * - เก็บข้อมูลใน localStorage (ไม่มีแบ็คเอนด์)
 * - นำเข้า/ส่งออกเป็น JSON ได้
 * 
 * หมายเหตุ: เพื่อความเรียบง่าย ไม่ใช้ไลบรารีเพิ่มเติม
 */

  // ตัวอย่างข้อมูลเล็ก ๆ สำหรับ 3 ตัวแรก

function today(offsetDays = 0) {
  const d = new Date();
  d.setDate(d.getDate() + offsetDays);
  return d.toISOString().slice(0, 10);
}
function addDays(days) {
  const d = new Date();
  d.setDate(d.getDate() + days);
  return d.toISOString().slice(0, 10);
}

const JOB_TYPES = [
  "ตรวจเช็ก",
  "บำรุงรักษาตามแผน",
  "ซ่อมฉุกเฉิน",
  "เปลี่ยนอะไหล่",
  "ทำความสะอาด",
];
const JOB_STATUS = ["ใหม่", "กำลังดำเนินการ", "เสร็จสิ้น", "รอดำเนินการ", "ยกเลิก"];
const PA_TYPES = ["มอเตอร์", "อะไหล่", "งานบริการ", "อื่นๆ"];
const PA_STATUS = ["ร่าง", "สั่งซื้อแล้ว", "รับของแล้ว", "ยกเลิก"];

export default function App() {
  const [data, setData] = useLocalState(LS_KEY, seedData());
  const [query, setQuery] = useState(""); // global search (ชื่อ/ประวัติพร้อมกัน)
  const [selectedCode, setSelectedCode] = useState("M001");
  const selectedMotor = useMemo(
    () => data.motors.find((m) => m.code === selectedCode) || data.motors[0],
    [data.motors, selectedCode]
  );

  const filteredMotors = useMemo(() => {
    const q = query.trim().toLowerCase();
    if (!q) return data.motors;
    return data.motors.filter((m) =>
      (m.code + " " + (m.name || "")).toLowerCase().includes(q)
    );
  }, [data.motors, query]);

  // Search history across all motors
  const historyResults = useMemo(() => {
    const q = query.trim().toLowerCase();
    if (!q) return [];
    return data.history
      .filter((h) =>
        [
          h.motorCode,
          h.date,
          h.type,
          h.issue,
          h.cause,
          h.fix,
          h.part,
          String(h.downtime ?? ""),
          h.owner,
          String(h.cost ?? ""),
          h.ref,
          h.pa,
          h.status,
          h.nextDue,
          h.notes,
        ]
          .join(" ")
          .toLowerCase()
          .includes(q)
      )
      .slice(0, 200); // cap results
  }, [data.history, query]);

  // Handlers: save motor info
  function updateMotorField(code, field, value) {
    setData((prev) => ({
      ...prev,
      motors: prev.motors.map((m) => (m.code === code ? { ...m, [field]: value } : m)),
    }));
  }

  // Handlers: history
  function addHistoryEntry(entry) {
    setData((prev) => ({ ...prev, history: [{ ...entry }, ...prev.history] }));
  }
  function deleteHistoryEntry(idx) {
    setData((prev) => ({
      ...prev,
      history: prev.history.filter((_, i) => i !== idx),
    }));
  }

  // Handlers: PA
  function addPAEntry(entry) {
    setData((prev) => ({ ...prev, paOrders: [{ ...entry }, ...prev.paOrders] }));
  }
  function deletePAEntry(idx) {
    setData((prev) => ({ ...prev, paOrders: prev.paOrders.filter((_, i) => i !== idx) }));
  }

  // Import/Export JSON
  function exportJSON() {
    const blob = new Blob([JSON.stringify(data, null, 2)], { type: "application/json" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `motor_data_${new Date().toISOString().slice(0, 10)}.json`;
    a.click();
    URL.revokeObjectURL(url);
  }
  function importJSON(ev) {
    const file = ev.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = () => {
      try {
        const obj = JSON.parse(String(reader.result));
        if (!obj.motors || !obj.history || !obj.paOrders) throw new Error("รูปแบบไฟล์ไม่ถูกต้อง");
        setData(obj);
      } catch (e) {
        alert("นำเข้าไม่สำเร็จ: " + e.message);
      }
    };
    reader.readAsText(file, "utf-8");
  }

  // Derived: history + PA for selected motor
  const motorHistory = useMemo(
    () => data.history.filter((h) => h.motorCode === selectedMotor?.code),
    [data.history, selectedMotor]
  );
  const motorPAs = useMemo(
    () => data.paOrders.filter((p) => p.motorCode === selectedMotor?.code),
    [data.paOrders, selectedMotor]
  );

  const [activeTab, setActiveTab] = useState("history"); // "history" | "pa"

  return (
    <div className="min-h-screen w-full bg-slate-50 text-slate-800">
      {/* Header */}
      <div className="sticky top-0 z-10 bg-white/90 backdrop-blur border-b border-slate-200">
        <div className="max-w-7xl mx-auto px-4 py-3 flex items-center gap-3">
          <div className="text-xl font-semibold">🛠️ ระบบบันทึกมอเตอร์ 100 ตัว</div>
          <div className="ml-auto flex items-center gap-2">
            <input
              value={query}
              onChange={(e) => setQuery(e.target.value)}
              placeholder="ค้นหาชื่อ/รหัส/ประวัติ/เลข PA/คำอธิบาย…"
              className="w-[420px] rounded-xl border border-slate-300 px-3 py-2 focus:outline-none focus:ring-2 focus:ring-sky-400"
            />
            <button
              onClick={() => setQuery("")}
              className="rounded-xl border px-3 py-2 hover:bg-slate-100"
            >ล้าง</button>
            <button onClick={exportJSON} className="rounded-xl border px-3 py-2 hover:bg-slate-100">ส่งออก JSON</button>
            <label className="rounded-xl border px-3 py-2 hover:bg-slate-100 cursor-pointer">
              นำเข้า JSON
              <input type="file" accept="application/json" onChange={importJSON} className="hidden" />
            </label>
          </div>
        </div>
        {/* Global search results (names + history) */}
        {query.trim() && (
          <div className="max-w-7xl mx-auto px-4 pb-3 grid grid-cols-1 md:grid-cols-2 gap-4">
            <div className="bg-white rounded-xl border p-3 shadow-sm">
              <div className="font-semibold mb-2">รายชื่อที่ตรง ({filteredMotors.length})</div>
              <div className="max-h-60 overflow-auto divide-y">
                {filteredMotors.slice(0, 200).map((m) => (
                  <div key={m.code} className="py-1 flex items-center justify-between">
                    <div>
                      <div className="font-medium">{m.code} – {m.name}</div>
                      <div className="text-xs text-slate-500">{m.location || "(ยังไม่ระบุที่ตั้ง)"}</div>
                    </div>
                    <button
                      className="text-sky-600 hover:underline"
                      onClick={() => setSelectedCode(m.code)}
                    >เปิด</button>
                  </div>
                ))}
              </div>
            </div>
            <div className="bg-white rounded-xl border p-3 shadow-sm">
              <div className="font-semibold mb-2">ประวัติที่ตรง ({historyResults.length})</div>
              <div className="max-h-60 overflow-auto divide-y">
                {historyResults.map((h, idx) => (
                  <div key={idx} className="py-1 text-sm">
                    <div className="flex items-center justify-between">
                      <div className="font-medium">{h.motorCode} · {h.date} · {h.type}</div>
                      <button className="text-sky-600 hover:underline" onClick={() => setSelectedCode(h.motorCode)}>ไปที่มอเตอร์</button>
                    </div>
                    <div className="text-slate-600">{h.issue || h.fix || h.notes}</div>
                    {h.pa && <div className="text-xs text-slate-500">PA: {h.pa} · เลขอ้างอิง: {h.ref}</div>}
                  </div>
                ))}
              </div>
            </div>
          </div>
        )}
      </div>

      {/* Body */}
      <div className="max-w-7xl mx-auto grid grid-cols-1 md:grid-cols-[320px,1fr] gap-4 px-4 py-4">
        {/* Sidebar: motor list */}
        <aside className="bg-white rounded-2xl border p-3 h-fit sticky top-[76px]">
          <div className="flex items-center gap-2 mb-2">
            <input
              value={query}
              onChange={(e) => setQuery(e.target.value)}
              placeholder="ค้นหารายชื่อ/รหัส…"
              className="w-full rounded-xl border border-slate-300 px-3 py-2 focus:outline-none focus:ring-2 focus:ring-sky-400"
            />
          </div>
          <div className="max-h-[65vh] overflow-auto divide-y">
            {filteredMotors.map((m) => (
              <button
                key={m.code}
                onClick={() => setSelectedCode(m.code)}
                className={`text-left w-full px-2 py-2 hover:bg-slate-50 ${
                  selectedMotor?.code === m.code ? "bg-sky-50 border-l-4 border-sky-400" : ""
                }`}
              >
                <div className="font-medium">{m.code} – {m.name}</div>
                <div className="text-xs text-slate-500">{m.location || "(ยังไม่ระบุที่ตั้ง)"}</div>
              </button>
            ))}
          </div>
        </aside>

        {/* Main: motor detail */}
        <main className="space-y-4">
          {/* Motor header */}
          <div className="bg-white rounded-2xl border p-4 shadow-sm">
            <div className="flex items-center gap-3">
              <div className="text-lg font-semibold">รายละเอียดมอเตอร์</div>
              <div className="ml-auto text-sm text-slate-500">รหัสคงที่ (เพื่อความถูกต้องของประวัติ)</div>
            </div>
            <div className="grid grid-cols-1 md:grid-cols-3 gap-3 mt-3">
              <LabeledInput label="รหัสมอเตอร์" value={selectedMotor?.code || ""} readOnly />
              <LabeledInput label="ชื่อมอเตอร์" value={selectedMotor?.name || ""} onChange={(v) => updateMotorField(selectedMotor.code, "name", v)} />
              <LabeledInput label="เลข PA มอเตอร์" value={selectedMotor?.pa || ""} onChange={(v) => updateMotorField(selectedMotor.code, "pa", v)} />
              <LabeledInput label="รุ่น/สเปค" value={selectedMotor?.spec || ""} onChange={(v) => updateMotorField(selectedMotor.code, "spec", v)} />
              <LabeledInput label="สถานที่" value={selectedMotor?.location || ""} onChange={(v) => updateMotorField(selectedMotor.code, "location", v)} />
              <LabeledInput type="date" label="วันที่ติดตั้ง" value={selectedMotor?.installed || ""} onChange={(v) => updateMotorField(selectedMotor.code, "installed", v)} />
              <LabeledInput label="ผู้ผลิต" value={selectedMotor?.manufacturer || ""} onChange={(v) => updateMotorField(selectedMotor.code, "manufacturer", v)} />
              <LabeledInput className="md:col-span-3" label="หมายเหตุ" value={selectedMotor?.notes || ""} onChange={(v) => updateMotorField(selectedMotor.code, "notes", v)} />
            </div>
          </div>

          {/* Tabs */}
          <div className="bg-white rounded-2xl border p-4 shadow-sm">
            <div className="flex items-center gap-2 border-b pb-2 mb-3">
              <TabButton active={activeTab === "history"} onClick={() => setActiveTab("history")}>ประวัติซ่อม/บำรุง</TabButton>
              <TabButton active={activeTab === "pa"} onClick={() => setActiveTab("pa")}>รายการ PA</TabButton>
            </div>

            {activeTab === "history" ? (
              <HistorySection
                motor={selectedMotor}
                rows={motorHistory}
                onAdd={addHistoryEntry}
                onDelete={(idx) => {
                  // ต้องหาค่า index ของ global history
                  const globalIdx = data.history.findIndex((h) => h === motorHistory[idx]);
                  if (globalIdx >= 0) deleteHistoryEntry(globalIdx);
                }}
              />
            ) : (
              <PASection
                motor={selectedMotor}
                rows={motorPAs}
                onAdd={addPAEntry}
                onDelete={(idx) => {
                  const globalIdx = data.paOrders.findIndex((p) => p === motorPAs[idx]);
                  if (globalIdx >= 0) deletePAEntry(globalIdx);
                }}
              />
            )}
          </div>
        </main>
      </div>

      {/* Footer */}
      <div className="text-center text-xs text-slate-500 py-6">ข้อมูลเก็บบนเครื่องของคุณ (localStorage) · ส่งออก/นำเข้า JSON ได้</div>
    </div>
  );
}

function TabButton({ active, onClick, children }) {
  return (
    <button
      onClick={onClick}
      className={`px-3 py-1.5 rounded-xl border ${active ? "bg-sky-600 text-white border-sky-600" : "bg-white hover:bg-slate-50"}`}
    >
      {children}
    </button>
  );
}

function LabeledInput({ label, value, onChange, type = "text", className = "", readOnly }) {
  return (
    <label className={`text-sm ${className}`}>
      <div className="mb-1 text-slate-600">{label}</div>
      <input
        type={type}
        value={value || ""}
        readOnly={readOnly}
        onChange={(e) => onChange && onChange(type === "number" ? Number(e.target.value) : e.target.value)}
        className={`w-full rounded-xl border border-slate-300 px-3 py-2 focus:outline-none focus:ring-2 focus:ring-sky-400 ${readOnly ? "bg-slate-100" : ""}`}
      />
    </label>
  );
}

function LabeledSelect({ label, value, onChange, options }) {
  return (
    <label className="text-sm">
      <div className="mb-1 text-slate-600">{label}</div>
      <select
        value={value || ""}
        onChange={(e) => onChange(e.target.value)}
        className="w-full rounded-xl border border-slate-300 px-3 py-2 bg-white focus:outline-none focus:ring-2 focus:ring-sky-400"
      >
        <option value="">— เลือก —</option>
        {options.map((op) => (
          <option key={op} value={op}>{op}</option>
        ))}
      </select>
    </label>
  );
}

function SectionTitle({ children }) {
  return <div className="font-semibold text-slate-700 mb-2">{children}</div>;
}

function HistorySection({ motor, rows, onAdd, onDelete }) {
  const [form, setForm] = useState({
    date: today(), type: "", issue: "", cause: "", fix: "", part: "", downtime: "", owner: "", cost: "", ref: "", pa: "", status: "", nextDue: "", notes: "",
  });

  function submit() {
    if (!motor?.code) return;
    if (!form.date || !form.type) {
      alert("กรอก 'วันที่' และ 'ประเภทงาน' อย่างน้อย");
      return;
    }
    onAdd({
      motorCode: motor.code,
      date: form.date,
      type: form.type,
      issue: form.issue,
      cause: form.cause,
      fix: form.fix,
      part: form.part,
      downtime: Number(form.downtime || 0),
      owner: form.owner,
      cost: Number(form.cost || 0),
      ref: form.ref,
      pa: form.pa,
      status: form.status,
      nextDue: form.nextDue,
      notes: form.notes,
    });
    setForm({ date: today(), type: "", issue: "", cause: "", fix: "", part: "", downtime: "", owner: "", cost: "", ref: "", pa: "", status: "", nextDue: "", notes: "" });
  }

  return (
    <div className="space-y-3">
      <SectionTitle>เพิ่มรายการใหม่</SectionTitle>
      <div className="grid grid-cols-1 md:grid-cols-4 gap-3">
        <LabeledInput type="date" label="วันที่" value={form.date} onChange={(v) => setForm({ ...form, date: v })} />
        <LabeledSelect label="ประเภทงาน" value={form.type} onChange={(v) => setForm({ ...form, type: v })} options={JOB_TYPES} />
        <LabeledSelect label="สถานะ" value={form.status} onChange={(v) => setForm({ ...form, status: v })} options={JOB_STATUS} />
        <LabeledInput label="เลขอ้างอิง" value={form.ref} onChange={(v) => setForm({ ...form, ref: v })} />
        <LabeledInput label="อาการ/ปัญหา" value={form.issue} onChange={(v) => setForm({ ...form, issue: v })} />
        <LabeledInput label="สาเหตุ" value={form.cause} onChange={(v) => setForm({ ...form, cause: v })} />
        <LabeledInput label="วิธีแก้ไข" value={form.fix} onChange={(v) => setForm({ ...form, fix: v })} />
        <LabeledInput label="อะไหล่ที่ใช้" value={form.part} onChange={(v) => setForm({ ...form, part: v })} />
        <LabeledInput type="number" label="เวลาเครื่องหยุด (ชม.)" value={form.downtime} onChange={(v) => setForm({ ...form, downtime: v })} />
        <LabeledInput label="ผู้รับผิดชอบ" value={form.owner} onChange={(v) => setForm({ ...form, owner: v })} />
        <LabeledInput type="number" label="ค่าใช้จ่าย (THB)" value={form.cost} onChange={(v) => setForm({ ...form, cost: v })} />
        <LabeledInput label="เลข PA" value={form.pa} onChange={(v) => setForm({ ...form, pa: v })} />
        <LabeledInput type="date" label="กำหนดบำรุงครั้งถัดไป" value={form.nextDue} onChange={(v) => setForm({ ...form, nextDue: v })} />
        <LabeledInput className="md:col-span-4" label="หมายเหตุ" value={form.notes} onChange={(v) => setForm({ ...form, notes: v })} />
      </div>
      <div className="flex gap-2">
        <button onClick={submit} className="px-4 py-2 rounded-xl bg-sky-600 text-white hover:bg-sky-700">บันทึกรายการ</button>
      </div>

      <SectionTitle>รายการล่าสุด</SectionTitle>
      <div className="overflow-auto rounded-xl border">
        <table className="min-w-full text-sm">
          <thead className="bg-slate-100">
            <tr>
              {["วันที่","ประเภทงาน","อาการ/ปัญหา","สาเหตุ","วิธีแก้ไข","อะไหล่","Downtime","ผู้รับผิดชอบ","ค่าใช้จ่าย","อ้างอิง","PA","สถานะ","ถัดไป","หมายเหตุ"," "].map((h) => (
                <th key={h} className="px-2 py-2 text-left whitespace-nowrap">{h}</th>
              ))}
            </tr>
          </thead>
          <tbody>
            {rows.map((r, i) => (
              <tr key={i} className="odd:bg-white even:bg-slate-50">
                <td className="px-2 py-1 whitespace-nowrap">{r.date}</td>
                <td className="px-2 py-1 whitespace-nowrap">{r.type}</td>
                <td className="px-2 py-1">{r.issue}</td>
                <td className="px-2 py-1">{r.cause}</td>

function PASection({ motor, rows, onAdd, onDelete }) {
  const [form, setForm] = useState({
    date: today(), paNo: "", paType: "", desc: "", qty: "", unit: "ชิ้น", unitPrice: "", vendor: "", status: "", invoice: "", link: "", notes: "",
  });

  function submit() {
    if (!motor?.code) return;
    if (!form.date || !form.paNo) {
      alert("กรอก 'วันที่สั่งซื้อ' และ 'เลข PA'");
      return;
    }
    onAdd({
      motorCode: motor.code,
      date: form.date,
      paNo: form.paNo,
      paType: form.paType,
      desc: form.desc,
      qty: Number(form.qty || 0),
      unit: form.unit,
      unitPrice: Number(form.unitPrice || 0),
      vendor: form.vendor,
      status: form.status,
      invoice: form.invoice,
      link: form.link,
      notes: form.notes,
    });
    setForm({ date: today(), paNo: "", paType: "", desc: "", qty: "", unit: "ชิ้น", unitPrice: "", vendor: "", status: "", invoice: "", link: "", notes: "" });
  }

  return (
    <div className="space-y-3">
      <SectionTitle>เพิ่ม PA</SectionTitle>
      <div className="grid grid-cols-1 md:grid-cols-4 gap-3">
        <LabeledInput type="date" label="วันที่สั่งซื้อ" value={form.date} onChange={(v) => setForm({ ...form, date: v })} />
        <LabeledInput label="เลข PA" value={form.paNo} onChange={(v) => setForm({ ...form, paNo: v })} />
        <LabeledSelect label="ประเภท PA" value={form.paType} onChange={(v) => setForm({ ...form, paType: v })} options={PA_TYPES} />
        <LabeledSelect label="สถานะ PA" value={form.status} onChange={(v) => setForm({ ...form, status: v })} options={PA_STATUS} />
        <LabeledInput className="md:col-span-2" label="คำอธิบาย" value={form.desc} onChange={(v) => setForm({ ...form, desc: v })} />
        <LabeledInput type="number" label="จำนวน" value={form.qty} onChange={(v) => setForm({ ...form, qty: v })} />
        <LabeledInput label="หน่วย" value={form.unit} onChange={(v) => setForm({ ...form, unit: v })} />
        <LabeledInput type="number" label="ราคาต่อหน่วย" value={form.unitPrice} onChange={(v) => setForm({ ...form, unitPrice: v })} />
        <LabeledInput label="ผู้ขาย" value={form.vendor} onChange={(v) => setForm({ ...form, vendor: v })} />
        <LabeledInput label="เลขใบกำกับ/ใบส่งของ" value={form.invoice} onChange={(v) => setForm({ ...form, invoice: v })} />
        <LabeledInput label="ลิงก์" value={form.link} onChange={(v) => setForm({ ...form, link: v })} />
        <LabeledInput className="md:col-span-4" label="หมายเหตุ" value={form.notes} onChange={(v) => setForm({ ...form, notes: v })} />
      </div>
      <div className="flex gap-2">
        <button onClick={submit} className="px-4 py-2 rounded-xl bg-sky-600 text-white hover:bg-sky-700">บันทึก PA</button>
      </div>

      <SectionTitle>รายการ PA ของมอเตอร์นี้</SectionTitle>
      <div className="overflow-auto rounded-xl border">
        <table className="min-w-full text-sm">
          <thead className="bg-slate-100">
            <tr>
              {["วันที่สั่งซื้อ","เลข PA","ประเภท","คำอธิบาย","จำนวน","หน่วย","ราคาต่อหน่วย","ราคารวม","ผู้ขาย","สถานะ","ใบกำกับ/ส่งของ","ลิงก์","หมายเหตุ"," "].map((h) => (
                <th key={h} className="px-2 py-2 text-left whitespace-nowrap">{h}</th>

            </tr>
          </thead>

            {rows.map((r, i) => (

                <td className="px-2 py-1 whitespace-nowrap">{r.date}</td>
                <td className="px-2 py-1 whitespace-nowrap">{r.paNo}</td>
                <td className="px-2 py-1 whitespace-nowrap">{r.paType}</td>
                <td className="px-2 py-1">{r.desc}</td>
                <td className="px-2 py-1 text-right whitespace-nowrap">{r.qty}</td>
                <td className="px-2 py-1 whitespace-nowrap">{r.unit}</td>
                <td className="px-2 py-1 text-right whitespace-nowrap">{r.unitPrice?.toLocaleString()}</td>
                <td className="px-2 py-1 text-right whitespace-nowrap">{(r.qty * r.unitPrice || 0).toLocaleString()}</td>
                <td className="px-2 py-1 whitespace-nowrap">{r.vendor}</td>

              </tr>
            ))}
            {rows.length === 0 && (
              <tr><td className="px-2 py-6 text-center text-slate-500" colSpan={14}>ยังไม่มีรายการ</td></tr>
            )}
          </tbody>
        </table>
      </div>
    </div>
  );
}

function useLocalState(key, initialValue) {
  const [state, setState] = useState(() => {
    try {
      const raw = localStorage.getItem(key);
      return raw ? JSON.parse(raw) : initialValue;
    } catch (e) {
      return initialValue;
    }
  });
  useEffect(() => {
    try {
      localStorage.setItem(key, JSON.stringify(state));
    } catch (e) {
      // ignore
    }
  }, [key, state]);
  return [state, setState];
}
